\begin{frame}

    \frametitle{mathematical content}

    \vspace{-0.5em}

    Given a dataset $\mathcal{D} = \{(\B{x}_i, \B{y}_i)\}_{i=1}^{N}$, we aim to:

    \vspace{1em}

    \begin{equation*}
        \min_{\Bs{\theta}} \frac{1}{N} \sum_{i=1}^{N} \mathcal{L}(f_{\Bs{\theta}}(\B{x}_i), \B{y}_i)
    \end{equation*}

    \vspace{1em}

    where:
    \begin{itemize}
        \setlength{\itemindent}{-1em}
        \setlength\itemsep{0.5em}
        \item $f_{\Bs{\theta}}$ is the model with parameters $\Bs{\theta}$
        \item $\mathcal{L}$ is the loss function
    \end{itemize}

\end{frame}
